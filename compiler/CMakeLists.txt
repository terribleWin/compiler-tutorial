cmake_minimum_required(VERSION 3.16)
project(sprout LANGUAGES CXX)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 11)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(sprout src/sprout.cpp)
if(MSVC)
  target_compile_options(sprout PRIVATE /W4 /utf-8)
else()
  target_compile_options(sprout PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
find_package(Python3 COMPONENTS Interpreter)
if(BUILD_TESTING AND Python3_Interpreter_FOUND)
  add_test(NAME sprout_suite COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_tests.py" "$<TARGET_FILE:sprout>")
endif()
